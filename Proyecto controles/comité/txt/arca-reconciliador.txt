    function mapRow(row, source) {
      const map = source === 'ARCA' ? COL_MAP.ARCA : COL_MAP.CDP;
      const mapped = {};
      for (const [key, val] of Object.entries(row)) {
        const normalized = map[key.trim().toLowerCase()] || key;
        mapped[normalized] = val;
      }

      let concatenado = mapped.concatenado || '';
      if (source === 'ARCA') {
        if (mapped.cuit !== undefined && mapped.punto_venta !== undefined && mapped.nro_desde !== undefined) {
          const cuitStr = String(mapped.cuit).replace(/\D/g, '').padStart(11, '0');
          const pvStr = String(mapped.punto_venta).replace(/\D/g, '').padStart(4, '0');
          const ndStr = String(mapped.nro_desde).replace(/\D/g, '').padStart(8, '0');
          concatenado = cuitStr + pvStr + ndStr;
        }
      } else if (source === 'CDP' || source === 'E1') {
        if (mapped.cuit !== undefined && mapped.factura !== undefined) {
          const cuitStr = String(mapped.cuit).replace(/\D/g, '').padStart(11, '0');
          let factStr = mapped.factura;
          if (typeof factStr === 'number') {
            factStr = factStr.toLocaleString('fullwide', { useGrouping: false });
          } else {
            factStr = String(factStr).trim();
          }
          factStr = factStr.replace(/\D/g, '').padStart(12, '0');
          concatenado = cuitStr + factStr;
        }
      }

      return {
        concatenado: concatenado,
        exchange_rate: mapped.exchange_rate || '',
        amount: parseAmount(mapped.amount),
        currency: normalizeCurrency(mapped.currency, source),
        vendor: mapped.vendor || '',
        issue_date: mapped.issue_date || '',
        posting_date: mapped.posting_date || '',
        doc_type: mapped.doc_type || '',
        erp: mapped.erp || source,
        duplicados: mapped.duplicados || false
      };
    }

    function processFile(file, type) {

          } else {

    function validateColumns(sampleRow, requiredNormalized) {
      const keys = Object.keys(sampleRow).map(k => String(k).trim().toLowerCase());
      return requiredNormalized.filter(req => !keys.includes(String(req).toLowerCase()));
    }

    function buildValidationPills(sampleRow, sourceKey) {
      const keys = Object.keys(sampleRow).map(k => String(k).trim().toLowerCase());
      const req = REQUIRED_COLS[sourceKey];
      const pills = req.map(col => {
        const found = keys.includes(String(col).toLowerCase());
        return `<span class="pill ${found ? 'ok' : 'missing'}">${found ? '✓' : '✗'} ${col}</span>`;
      });
      return `<div class="validation-pills">${pills.join('')}</div>`;
    }

    // ══════════════════════════════════════════════════════
    //  NORMALIZATION
    // ══════════════════════════════════════════════════════
    function normalizeCurrency(currency, source) {
      if (!currency) return source === 'CDP' || source === 'E1' ? DEFAULT_ERP_CURRENCY : null;
      const s = String(currency).trim().toUpperCase();
      if (source === 'ARCA' && CURRENCY_MAP.ARCA[s]) return CURRENCY_MAP.ARCA[s];
      if (CURRENCY_MAP.genericas[s]) return CURRENCY_MAP.genericas[s];
      return s;
    }

    function parseAmount(val) {
      if (val === null || val === undefined) return 0;
      if (typeof val === 'number') return isNaN(val) ? 0 : val;
      let s = String(val).replace(/[^\d.,-]/g, '').trim();
      if (!s) return 0;
      if (s.includes(',') && s.includes('.')) {
        s = s.lastIndexOf(',') > s.lastIndexOf('.')
          ? s.replace(/\./g, '').replace(',', '.')
          : s.replace(/,/g, '');
      } else if (s.includes(',')) {
        // FIX: Contexto Argentina → coma decimal, punto separador de miles
        // Si solo hay coma, tratarla como decimal (ej: "1500,50" → 1500.50)
        // Solo tratar como separador de miles si hay exactamente 3 dígitos después Y hay más dígitos antes
        const parts = s.split(',');
        if (parts.length === 2 && parts[1].length === 3 && parts[0].length > 3) {
          s = s.replace(',', ''); // separador de miles: "1,500" → "1500"
        } else {
          s = s.replace(',', '.'); // decimal: "1500,50" → "1500.50"
        }
      }
      const n = parseFloat(s);
      return isNaN(n) ? 0 : n;
    }

    function parseDate(val) {
      if (!val) return null;
      if (val instanceof Date) return val;
      const s = String(val).trim();
      let d = new Date(s);
      if (isNaN(d.getTime())) {
        const num = parseFloat(s);
        if (!isNaN(num) && num > 0) {
          d = new Date((num - 25569) * 86400 * 1000);
        } else return null;
      }
      return isNaN(d.getTime()) ? null : d;
    }

    function reconcile() {
      if (state.processing) return;
      state.processing = true;
      showProgress(true, 'Normalizando datos…', 10);
      document.getElementById('statsSection').style.display = '';

      setTimeout(() => {
        try {
          const tolValue = parseFloat(document.getElementById('tolValue').value) || 0;
          const tolType = document.getElementById('tolType').value;

          const arcaRows = state.arcaData.map(r => mapRow(r, 'ARCA'));
          showProgress(true, 'Procesando ERP…', 30);

          let erpRows = [];
          if (state.erpData) erpRows = [...erpRows, ...state.erpData.map(r => mapRow(r, 'CDP'))];
          if (state.e1Data) erpRows = [...erpRows, ...state.e1Data.map(r => mapRow(r, 'E1'))];

          const postingDates = [];
          for (const row of erpRows) {
            if (row.posting_date) {
              const d = parseDate(row.posting_date);
              if (d) postingDates.push(d);
            }
          }
          if (postingDates.length) {
            postingDates.sort((a, b) => b - a);
            state.lastPostingDate = postingDates[0];
          }

          const selectedDate = document.getElementById('postingDateSelect')?.value;
          if (document.getElementById('enablePostingFilter')?.checked && selectedDate) {
            erpRows = erpRows.filter(row => {
              if (!row.posting_date) return false;
              const d = parseDate(row.posting_date);
              return d ? dateToString(d) === selectedDate : false;
            });
          }

          showProgress(true, 'Cruzando registros…', 60);

          const erpIndex = new Map();
          for (const row of erpRows) {
            if (!row.concatenado) continue;
            if (!erpIndex.has(row.concatenado)) erpIndex.set(row.concatenado, []);
            erpIndex.get(row.concatenado).push(row);
          }

          const results = [];

          for (const arcaRow of arcaRows) {
            const key = arcaRow.concatenado;
            if (!key) {
              results.push({ ...arcaRow, status: 'Sin Concatenado', severity: 'warn' });
              continue;
            }

            const matches = erpIndex.get(key);

            // Si no hay match en ERP simplemente se omite (el ERP tiene rango acotado de fechas)
            if (!matches || !matches.length) continue;

            const match = matches[0];

            const amtDiff = Math.abs(arcaRow.amount - match.amount);
            const tolerance = tolType === 'pct'
              ? (arcaRow.amount * tolValue / 100)
              : tolValue;

            let status = 'OK', severity = 'ok';

            const arcaTc = Number(arcaRow.exchange_rate) || 0;
            const matchTc = Number(match.exchange_rate) || 0;
            let tcError = false;

            if (arcaTc >= 2 && matchTc >= 2) {
              if (arcaTc >= 1300 && arcaTc <= 1500) {
                if (Math.abs(arcaTc - matchTc) > 100) tcError = true;
              } else {
                if (arcaTc !== matchTc) tcError = true;
              }
            }

            if (tcError) { status = 'Error de Tipo Cambio'; severity = 'error'; }
            else if (arcaRow.currency !== match.currency) { status = 'Error de Moneda'; severity = 'error'; }
            else if (amtDiff > tolerance) { status = 'Error de Monto'; severity = 'error'; }

            results.push({ ...arcaRow, erpMatch: match, status, severity, amtDiff });
          }

          // Detectar duplicados entre distintos ERPs (misma factura en CDP + E1, etc.)
          // Agrupar todas las filas del ERP por concatenado y ver si hay más de un ERP distinto
          const erpByConcat = new Map();
          for (const row of erpRows) {
            if (!row.concatenado) continue;
            if (!erpByConcat.has(row.concatenado)) erpByConcat.set(row.concatenado, new Set());
            erpByConcat.get(row.concatenado).add(row.erp || 'ERP');
          }
          // Marcar resultados cuya factura aparece en más de un ERP distinto
          for (const r of results) {
            if (!r.concatenado) continue;
            const erps = erpByConcat.get(r.concatenado);
            if (erps && erps.size > 1) {
              r.status = `Duplicado entre ERPs (${[...erps].join(' + ')})`;
              r.severity = 'warn';
              r.crossErpSystems = [...erps];
            }
          }

          state.results = results;

          showProgress(true, 'Calculando estadísticas…', 90);
          const stats = computeStats(results);
          renderStats(stats);
          showProgress(false);

          document.getElementById('exportBtn').disabled = false;

          addMessage('agent', buildSummaryHtml(stats));
          state.conversationHistory = [];
          state.processing = false;
        } catch (err) {
          showProgress(false);
          state.processing = false; // FIX: siempre se resetea aunque falle
          addMessage('agent', `<div class="error-alert">❌ Error en control de integridad y duplicados: ${escHtml(err.message)}</div>`);
        }
      }, 50);
    }

    function computeStats(results) {
      const total = results.length;
      const ok = results.filter(r => r.severity === 'ok').length;
      const errors = results.filter(r => r.severity === 'error').length;
      const warnings = results.filter(r => r.severity === 'warn').length;
      const amtErr = results.filter(r => r.status === 'Error de Monto').length;
      const curErr = results.filter(r => r.status === 'Error de Moneda').length;
      const tcErr = results.filter(r => r.status === 'Error de Tipo Cambio').length;
      const crossErpDup = results.filter(r => r.status?.startsWith('Duplicado entre ERPs')).length;
      return { total, ok, errors, warnings, amtErr, curErr, tcErr, crossErpDup };
    }

    function renderStats(s) {

    <div class="stat-card"><div class="stat-label">OK</div><div class="stat-value ok">${s.ok} <span style="font-size:12px;color:var(--text3)">(${pctFn(s.ok)}%)</span></div></div>
    <div class="stat-card"><div class="stat-label">Error Monto</div><div class="stat-value error">${s.amtErr}</div></div>

    <div class="stat-card"><div class="stat-label">Error T/C</div><div class="stat-value error">${s.tcErr}</div></div>
    <div class="stat-card"><div class="stat-label">Dup. entre ERPs</div><div class="stat-value warn">${s.crossErpDup}</div></div>
  `;
    }

    // ══════════════════════════════════════════════════════
    //  AI CALL — con retry logic, timeout y manejo correcto de historial
    // ══════════════════════════════════════════════════════
    async function callAI(userMessage, attempt = 0) {
      const cfg = PROVIDERS[state.provider];
      const contextSummary = buildContextSummary();

      // FIX: NO agregar el turno de usuario al historial hasta confirmar respuesta exitosa
      const MAX_HISTORY = 8;
      const history = state.conversationHistory.slice(-MAX_HISTORY);

      const systemPrompt = `Sos un auditor experto en facturas de proveedores para empresas argentinas.
Tenés acceso a los resultados de un control de integridad y duplicados entre ARCA/AFIP y los sistemas ERP (CDP, SAP, E1).
Respondé siempre en español, de forma concisa y profesional.
Si hay datos disponibles, usá los números exactos.
No inventes datos que no estén en el contexto.

CONTEXTO DE CONTROL DE INTEGRIDAD Y DUPLICADOS:
${contextSummary}`;

      const messages = [...history, { role: 'user', content: userMessage }];

      const timeoutPromise = new Promise((_, reject) =>
        setTimeout(() => reject(new Error('La solicitud tardó demasiado (timeout 30s). Intentá de nuevo.')), API_TIMEOUT_MS)
      );

      let url = '';
      let headers = { 'Content-Type': 'application/json' };
      let bodyObj = {};

      if (state.provider === 'gemini') {
        url = `https://generativelanguage.googleapis.com/v1beta/models/${state.model}:generateContent?key=${state.apiKey}`;
        bodyObj = {
          system_instruction: { parts: [{ text: systemPrompt }] },
          contents: messages.map(m => ({
            role: m.role === 'assistant' ? 'model' : 'user',
            parts: [{ text: m.content }]
          })),
          generationConfig: { maxOutputTokens: 800 }
        };
      } else if (state.provider === 'openai') {
        const openaiRequest = {
          model: state.model,
          messages: [
            { role: 'system', content: systemPrompt },
            ...messages.map(m => ({ role: m.role, content: m.content }))
          ],
          max_tokens: 800
        };

        if (state.useProxy) {
          url = state.proxyUrl;
          bodyObj = {
            targetEndpoint: 'https://api.openai.com/v1/chat/completions',
            provider: 'OpenAI',
            apiKey: state.apiKey,
            request: openaiRequest
          };
        } else {
          url = 'https://api.openai.com/v1/chat/completions';
          headers['Authorization'] = `Bearer ${state.apiKey}`;
          bodyObj = openaiRequest;
        }
      }

      const fetchPromise = fetch(url, {
        method: 'POST',
        headers: headers,
        body: JSON.stringify(bodyObj)
      });

      try {
        const response = await Promise.race([fetchPromise, timeoutPromise]);

        if (!response.ok) {
          const errBody = await response.json().catch(() => ({}));
          const errMsg = errBody?.error?.message || `HTTP ${response.status}`;

          if (response.status === 429 && attempt < MAX_RETRIES) {
            const wait = (attempt + 1) * 2000;
            showToast(`⏳ Rate limit — reintentando en ${wait / 1000}s…`, 'warn');
            await sleep(wait);
            return callAI(userMessage, attempt + 1);
          }
          throw new Error(errMsg);
        }

        const data = await response.json();
        let reply = '';
        if (state.provider === 'gemini') {
          reply = data.candidates?.[0]?.content?.parts?.[0]?.text || '(sin respuesta)';
        } else if (state.provider === 'openai') {
          reply = data.choices?.[0]?.message?.content || '(sin respuesta)';
        }

        // Solo agregar al historial si la respuesta fue exitosa
        state.conversationHistory.push({ role: 'user', content: userMessage });
        state.conversationHistory.push({ role: 'assistant', content: reply });

        // Mantener historial acotado
        if (state.conversationHistory.length > MAX_HISTORY) {
          state.conversationHistory = state.conversationHistory.slice(-MAX_HISTORY);
        }

        return reply;

      } catch (err) {
        throw err;
      }
    }

    function buildContextSummary() {
      if (!state.results) return 'No hay datos cargados aún.';

      const stats = computeStats(state.results);
      let summary = `Total facturas ARCA: ${stats.total}\n`;
      summary += `OK: ${stats.ok} | Errores monto: ${stats.amtErr} | Errores moneda: ${stats.curErr}\n`;
      summary += `Duplicados entre ERPs: ${stats.crossErpDup}\n\n`;

      const errorRows = state.results.filter(r => r.severity !== 'ok').slice(0, MAX_CONTEXT_ROWS);
      if (errorRows.length) {
        summary += 'Muestra de registros con problemas:\n';
        for (const r of errorRows.slice(0, 30)) {
          summary += `- [${r.status}] Concat: ${r.concatenado || 'N/A'} | Monto: ${r.amount} ${r.currency || ''}\n`;
        }
        if (errorRows.length > 30) summary += `...y ${errorRows.length - 30} más\n`;
      }

      return summary;
    }

    // ══════════════════════════════════════════════════════
    function addMessage(role, html, withRetry = false) {
      const container = document.getElementById('messages');
      const div = document.createElement('div');
      div.className = `msg ${role}`;

      const avatar = document.createElement('div');
      avatar.className = `avatar ${role}`;
      avatar.textContent = role === 'agent' ? '🧾' : '👤';

      const bubble = document.createElement('div');
      bubble.className = `bubble ${role}`;
      bubble.innerHTML = html;

      if (withRetry && role === 'agent') {
        const retryWrap = document.createElement('div');
        retryWrap.className = 'error-alert';
        retryWrap.innerHTML = `❌ ${html} <button class="retry-btn" onclick="retryLast()">↺ Reintentar</button>`;
        bubble.innerHTML = '';
        bubble.appendChild(retryWrap);
      }

      div.appendChild(avatar);
      div.appendChild(bubble);
      container.appendChild(div);
      container.scrollTop = container.scrollHeight;
      return bubble;
    }

    function showTyping() {
      const container = document.getElementById('messages');
      const div = document.createElement('div');
      div.className = 'msg agent';
      div.id = 'typingMsg';
      div.innerHTML = `
    <div class="avatar agent">🧾</div>
    <div class="bubble agent">
      <div class="typing-indicator"><span></span><span></span><span></span></div>
    </div>`;
      container.appendChild(div);
      container.scrollTop = container.scrollHeight;
    }

    function removeTyping() {
      document.getElementById('typingMsg')?.remove();
    }

    async function sendMessage() {
      const textarea = document.getElementById('userInput');
      const text = textarea.value.trim();
      if (!text || state.processing) return;
      if (!state.apiKey) { showToast('❌ Conectá una API key primero', 'error'); return; }
      if (text.length > MAX_INPUT_CHARS) {
        showToast(`❌ Mensaje muy largo (máx. ${MAX_INPUT_CHARS} chars)`, 'error'); return;
      }

      state.lastUserMessage = text;
      textarea.value = '';
      updateCharCount(textarea);
      addMessage('user', escHtml(text));

      document.getElementById('sendBtn').disabled = true;
      showTyping();

      try {
        const reply = await callAI(text);
        removeTyping();
        addMessage('agent', escHtml(reply).replace(/\n/g, '<br>'));
      } catch (err) {
        removeTyping();
        addMessage('agent', `❌ Error: ${escHtml(err.message)}`, true);
        showToast('❌ ' + err.message, 'error');
      } finally {
        document.getElementById('sendBtn').disabled = false;
      }
    }

    // FIX: retryLast elimina correctamente los últimos 2 turnos (usuario + asistente fallido)
    function retryLast() {
      if (!state.lastUserMessage) return;
      // Eliminar los últimos 2 turnos si existen (pueden no existir si falló antes del push)
      if (state.conversationHistory.length >= 2) {
        state.conversationHistory = state.conversationHistory.slice(0, -2);
      }
      const msgs = document.getElementById('messages');
      msgs.lastElementChild?.remove();

      document.getElementById('userInput').value = state.lastUserMessage;
      sendMessage();
    }

    function ask(text) {
      document.getElementById('userInput').value = text;
      sendMessage();
    }

    function handleKey(e) {
      if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendMessage(); }
    }

    function updateCharCount(textarea) {
      const len = textarea.value.length;
      const counter = document.getElementById('charCount');
      if (len > MAX_INPUT_CHARS * 0.8) {
        counter.style.display = '';
        counter.textContent = `${len} / ${MAX_INPUT_CHARS}`;
        counter.style.color = len > MAX_INPUT_CHARS ? 'var(--danger)' : 'var(--warn)';
      } else {
        counter.style.display = 'none';
      }
      textarea.style.height = 'auto';
      textarea.style.height = Math.min(textarea.scrollHeight, 120) + 'px';
    }

        [`${stats.crossErpDup} facturas aparecen duplicadas en más de un ERP.`]
      ];
      XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(summaryData), 'Resumen');

          'Fecha Posting': postingDate,
          'Estado': r.status,
          'Severidad': r.severity,
          'Nro de Factura': r.concatenado || '',
          'Nro Doc': r.erpMatch?.doc_type || '',
          'Proveedor': r.erpMatch?.vendor || '',

          'Moneda CDP': r.erpMatch?.currency || '',
          'Monto CDP': r.erpMatch?.amount ?? '',
          'Diferencia': r.amtDiff ?? '',

        };

      showToast('📥 Excel descargado', 'success');
    }

    function escHtml(str) {
      return String(str ?? '')
        .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;').replace(/'/g, '&#039;');
    }

    function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

    function pct(v, total) { return total ? `${Math.round(v / total * 100)}%` : '0%'; }

    function datestamp() {
      return new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
    }

    let toastTimeout;
    function showToast(msg, type = '') {
      const t = document.getElementById('toast');
      t.textContent = msg;
      t.className = `toast show ${type}`;
      clearTimeout(toastTimeout);
      toastTimeout = setTimeout(() => t.className = 'toast', 3500);
    }

    function showProgress(visible, label = '', pctVal = 0) {
      const wrap = document.getElementById('progressWrap');
      const bar = document.getElementById('progressBar');
      const lbl = document.getElementById('progressLabel');
      if (visible) {
        wrap.style.display = '';
        bar.style.width = pctVal + '%';
        lbl.textContent = label;
      } else {
        bar.style.width = '100%';
        setTimeout(() => { wrap.style.display = 'none'; bar.style.width = '0%'; }, 600);
      }
    }
